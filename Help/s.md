**ARM64 ассемблер — шпаргалка:**

## 1. Регистры
- `x0-x28` — 64-битные общие
- `w0-w28` — 32-битные (младшая часть x)
- `sp` — указатель стека
- `lr` (x30) — адрес возврата
- `pc` — счётчик команд

## 2. Загрузка/сохранение
```assembly
ldr x0, [x1]        // x0 = *x1 (8 байт)
ldr w0, [x1]        // w0 = *x1 (4 байта)
ldrb w0, [x1]       // w0 = *x1 (1 байт)
str x0, [x1]        // *x1 = x0
strb w0, [x1]       // *x1 = w0 (1 байт)
```

## 3. Арифметика
```assembly
add x0, x1, x2      // x0 = x1 + x2
add x0, x1, #5      // x0 = x1 + 5
sub x0, x1, x2      // x0 = x1 - x2
mul x0, x1, x2      // x0 = x1 × x2
udiv x0, x1, x2     // x0 = x1 ÷ x2 (беззнак.)
```

## 4. Логические
```assembly
and x0, x1, x2      // x0 = x1 & x2
orr x0, x1, x2      // x0 = x1 | x2
eor x0, x1, x2      // x0 = x1 ^ x2
lsl x0, x1, #2      // x0 = x1 << 2 (×4)
lsr x0, x1, #2      // x0 = x1 >> 2 (÷4)
```

## 5. основные флаги в ARM64
```assembly
Z (Zero) = 1 если результат равен нулю
N (Negative) = 1 если результат отрицательный
C (Carry) = 1 если было переполнение
V (oVerflow) = 1 если арифметическое переполнение
```
## 6. Сравнения и переходы
```assembly
tst x0, #0x20      // сравнить x0 и бит #0x20 (логическое побитовое И, устанавливает флаги)
cmp x0, x1          // сравнить x0 и x1 (устанавливает флаги)
b.eq label          // переход если =
b.ne label          // переход если ≠
b.gt label          // переход если >
b.lt label          // переход если <
b.ge label          // переход если ≥
b.le label          // переход если ≤
bl func             // вызов функции
ret                 // возврат из функции
1:                  // метка
1b                  //перейти к метке 1 (искать выше) 
1f                  //перейти к метке 1 (искать ниже)
(b или f нужно для цифровых меток)
```

## 7. Условия
```assembly
csel x0, x1, x2, eq // if (eq) x0=x1 else x0=x2
cset x0, eq         // if (eq) x0=1 else x0=0
```

## 8. Стек
```assembly
stp x0, x1, [sp, #-16]! // push x0,x1
ldp x0, x1, [sp], #16   // pop x0,x1
```

## 9. Константы
```assembly
mov x0, #42         // непосредственное значение
movz x0, #0x1234, lsl #16 // 0x12340000
ldr x0, =label      // загрузить адрес метки
```

## 10. Комментарии
```assembly

    комментарий */
```

## 11. Директивы
```assembly
.global name        // экспорт символа
.align 4            // выравнивание 16 байт
.text               // секция кода
.data               // секция данных
.word 0x12345678    // 32-битное слово
.quad 0x12345678ABCDEF12 // 64-битное
.ascii "string"     // строка без нуля
.asciz "string"     // строка с нулём
.space 1024         // 1024 нулевых байта
```
